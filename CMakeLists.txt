function(bellwether_dependency name repository revision localName)
  if(BELLWETHER_USE_LOCAL_VENDOR AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/Vendor/${localName}/.git")
    execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/Vendor/${localName}"
      OUTPUT_VARIABLE actualRevision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    if(NOT actualRevision STREQUAL revision)
      message(FATAL_ERROR "${name}: local dependency revision does not match lock")
    endif()
    execute_process(COMMAND git diff --quiet HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/Vendor/${localName}"
      RESULT_VARIABLE dependencyModified)
    if(NOT dependencyModified EQUAL 0)
      message(FATAL_ERROR "${name}: workspace dependency has unreviewed modifications")
    endif()
    set(FETCHCONTENT_SOURCE_DIR_${name} "${CMAKE_CURRENT_SOURCE_DIR}/Vendor/${localName}" CACHE PATH "" FORCE)
  endif()
  string(TOLOWER "${name}" lowerName)
  FetchContent_Declare(${name}
    DOWNLOAD_COMMAND ${CMAKE_COMMAND} "-DDEPENDENCY_REPOSITORY=${repository}" "-DDEPENDENCY_REVISION=${revision}"
      "-DDEPENDENCY_DIRECTORY=${FETCHCONTENT_BASE_DIR}/${lowerName}-src" -P "${CMAKE_CURRENT_SOURCE_DIR}/Tools/FetchDependency.cmake"
    UPDATE_COMMAND "")
endfunction()
